[Unit]
Description=Minimal RISC-V Weston desktop
After=systemd-user-sessions.service systemd-udev-trigger.service seatd.service
Requires=seatd.service
Conflicts=getty@tty1.service
[Service]
Type=exec
User=desktop
Group=desktop
SupplementaryGroups=video input render audio
WorkingDirectory=/home/desktop
RuntimeDirectory=replay-desktop
RuntimeDirectoryMode=0700
Environment=XDG_RUNTIME_DIR=/run/replay-desktop
Environment=WAYLAND_DISPLAY=wayland-0
Environment=LIBSEAT_BACKEND=seatd
Environment=XDG_SESSION_TYPE=wayland
Environment=XDG_CURRENT_DESKTOP=weston
TTYPath=/dev/tty1
StandardInput=tty
StandardOutput=journal+console
StandardError=journal+console
TTYReset=yes
TTYVHangup=yes
TTYVTDisallocate=yes
ExecStart=/usr/bin/dbus-run-session -- /usr/local/bin/replay-desktop
Restart=on-failure
RestartSec=3
[Install]
WantedBy=multi-user.target
